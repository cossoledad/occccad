find_package(GTest REQUIRED CONFIG)
include(GoogleTest)

add_executable(occccad_plane_gcs_adapter_test
    plane_gcs_sketch_solver_test.cpp
)
target_link_libraries(occccad_plane_gcs_adapter_test
    PRIVATE occccad_sketch_solver GTest::gtest_main
)
target_compile_features(occccad_plane_gcs_adapter_test PRIVATE cxx_std_17)
occccad_set_warnings(occccad_plane_gcs_adapter_test PRIVATE)
occccad_apply_sanitizers(occccad_plane_gcs_adapter_test)
gtest_discover_tests(occccad_plane_gcs_adapter_test)

add_executable(occccad_geometry_exchange_test
    geometry_exchange_test.cpp
)

    PRIVATE occccad_occt_kernel occccad_kernel_api GTest::gtest_main
)
target_compile_definitions(occccad_geometry_exchange_test
    PRIVATE OCCCCAD_MODEL_FIXTURE_DIR="${PROJECT_SOURCE_DIR}/models"
)

occccad_set_warnings(occccad_geometry_exchange_test PRIVATE)
occccad_apply_sanitizers(occccad_geometry_exchange_test)
gtest_discover_tests(occccad_geometry_exchange_test)
